Node class
Every state in the search space is wrapped in a Node. It stores the state itself, a pointer to its parent (for path reconstruction), and three cost values: g (cost so far), h (heuristic estimate), and f = g + h (total estimated cost used by A*). 
class Node:
    def __init__(self, state, parent=None, g=0, h=0):
        self.state  = state    # the current state (tuple, list, int, …)
        self.parent = parent   # parent Node, or None for the root
        self.g = g             # cost from start → this node
        self.h = h             # heuristic estimate: this node → goal
        self.f = g + h         # total estimated cost  (used by A*)

    def path(self):
        """Reconstruct the path from root to this node."""
        node, p = self, []
        while node:
            p.append(node.state)
            node = node.parent
        return p[::-1]          # reverse → start first

    def __lt__(self, other):   # needed so heapq can compare Nodes
        return self.f < other.f
Unified search() — one function, five algorithms
All five algorithms share the same logic. The only difference is what key the frontier is sorted by:
BFS → FIFO queue  ·  DFS → LIFO stack  ·  UCS → min-heap on g  ·  Best-First → min-heap on h  ·  A* → min-heap on f = g + h 
import heapq
from collections import deque

def search(start, goal_fn, expand_fn, h_fn, algo):
    """
    Generic search engine.

    Parameters
    ----------
    start     : initial state
    goal_fn   : state -> bool          (True when solved)
    expand_fn : node  -> [(state, cost), ...]
    h_fn      : state -> float         (heuristic; return 0 for uninformed)
    algo      : 'bfs' | 'dfs' | 'ucs' | 'bestfs' | 'astar'

    Returns
    -------
    dict  { 'path': [states] or None,
            'visited': [states in expansion order],
            'expanded': int }
    """
    root = Node(start, h=h_fn(start))

    # ── FRONTIER — the only line that differs per algorithm ─────
    if   algo == 'bfs':    frontier = deque([root])
    elif algo == 'dfs':    frontier = [root]                  # list used as stack
    elif algo == 'ucs':    frontier = [(root.g, 0, root)]      # key = g
    elif algo == 'bestfs': frontier = [(root.h, 0, root)]      # key = h
    elif algo == 'astar':  frontier = [(root.f, 0, root)]      # key = f
    # ─────────────────────────────────────────────────────────────

    visited   = set()
    ctr       = 0        # tie-breaker for heapq
    visit_ord = []        # order nodes were first expanded

    while frontier:

        # ── POP (also differs per algorithm) ─────────────────────────
        if   algo == 'bfs': node = frontier.popleft()
        elif algo == 'dfs': node = frontier.pop()
        else:               _, _, node = heapq.heappop(frontier)
        # ─────────────────────────────────────────────────────────────

        key = str(node.state)
        if key in visited:
            continue
        visited.add(key)
        visit_ord.append(node.state)
        ctr += 1

        if goal_fn(node.state):
            return {'path': node.path(), 'visited': visit_ord, 'expanded': ctr}

        for next_state, step_cost in expand_fn(node):
            if str(next_state) not in visited:
                ng    = node.g + step_cost
                nh    = h_fn(next_state)
                child = Node(next_state, node, ng, nh)

                # ── PUSH ─────────────────────────────────────────────────
                if   algo == 'bfs':    frontier.append(child)
                elif algo == 'dfs':    frontier.append(child)
                elif algo == 'ucs':    heapq.heappush(frontier, (child.g, ctr, child))
                elif algo == 'bestfs': heapq.heappush(frontier, (child.h, ctr, child))
                elif algo == 'astar':  heapq.heappush(frontier, (child.f, ctr, child))

    return {'path': None, 'visited': visit_ord, 'expanded': ctr}
Problem formulation
State: a single node ID (integer).
Actions: follow any edge from the current node to a neighbour.
Step cost: the weight on that edge.
Goal test: current node ID == goal node ID.
Heuristic (Best-First / A*): straight-line (Euclidean) distance between the current node's canvas position and the goal node's canvas position.

# Euclidean (straight-line) distance is always ≤ actual road distance
# → never overestimates → ADMISSIBLE → A* finds optimal path.
def h_fn(state):
    x1, y1 = coords[state]   # canvas position of current node
    x2, y2 = coords[GOAL]    # canvas position of goal node
    return math.hypot(x2-x1, y2-y1)   # straight-line distance

Problem formulation
State: (row, col) tuple — the agent's current cell.
Actions: move Up, Down, Left, Right (4-directional). Each action is only valid if the target cell is inside the grid and not a wall.
Step cost: the traversal cost of the destination cell (default 1; custom costs shown in gold).
Goal test: state == goal.
Heuristic (Best-First / A*): Manhattan distance — |row - goal_row| + |col - goal_col|. This is admissible because you can never reach the goal in fewer steps than the Manhattan distance on a grid.

# ── Grid definition ───────────────────────────────────────────
# '.' = open, '#' = wall, 'S' = start, 'G' = goal
GRID = [
    list("S.........."),
    list(".#####.####"),
    list(".#...#.#..."),
    list(".#.#.#.#.#."),
    list(".#.#...#.#."),
    list(".#.#####.#."),
    list("...#.....#G"),
]
ROWS = len(GRID)
COLS = len(GRID[0])
START = next((int(r),int(c)) for r in range(ROWS) for c in range(COLS) if GRID[r][c]=='S')
GOAL  = next((int(r),int(c)) for r in range(ROWS) for c in range(COLS) if GRID[r][c]=='G')

# Optional: custom costs per cell (row, col) -> cost
COSTS = {(2,2): 3, (3,4): 5}   # leave empty {} for uniform cost

# ── Problem functions ─────────────────────────────────────────
DIRS = [(-1,0), (1,0), (0,-1), (0,1)]   # Up, Down, Left, Right

# goal_fn: state is a (row, col) tuple.
# Return True when the agent has reached the goal cell.
def goal_fn(state):
    return state == GOAL   # e.g. state==(6,10) and GOAL==(6,10)

# expand_fn: Try all 4 directions from the current cell.
# Skip cells that are outside the grid or are walls ('#').
# Return valid (next_cell, step_cost) pairs.
def expand_fn(node):
    r, c = node.state          # unpack current (row, col)
    nbrs = []
    for dr, dc in DIRS:
        nr, nc = r+dr, c+dc   # candidate next cell
        # Check 1: inside the grid boundaries
        if 0 <= nr < ROWS and 0 <= nc < COLS:
            # Check 2: not a wall
            if GRID[nr][nc] != '#':
                cost = COSTS.get((nr, nc), 1)   # custom cost or default 1
                nbrs.append(((nr, nc), cost))
    return nbrs

# h_fn: Manhattan distance from current cell to goal.
# = (how many rows apart) + (how many columns apart).
# Always ≤ actual path length (walls may force a longer route)
# → ADMISSIBLE → A* is guaranteed to find the shortest path.
def h_fn(state):
    row_dist = abs(state[0] - GOAL[0])   # vertical distance
    col_dist = abs(state[1] - GOAL[1])   # horizontal distance
    return row_dist + col_dist

Problem formulation
State: a partial list of queen row-positions — one entry per column placed so far. [2, 0] means column 0 has a queen in row 2, column 1 has a queen in row 0.
Initial state: [] — no queens placed yet.
Actions: place one queen in any safe row of the next column.
Constraint (safe row): the new queen must not share a row or diagonal with any existing queen.
Goal test: all 8 columns are filled (len(state) == 8).
Heuristic (Best-First / A*): number of attacking pairs among the queens placed so far. Zero means no conflicts — we want to minimise this.

# ── Problem functions ─────────────────────────────────────────
N = 8   # board size (8 queens on an 8×8 board)

def is_safe(placement, col, row):
    """Can we place a queen at (row, col) given existing placement?"""
    for c, r in enumerate(placement):   # c = already-filled column, r = its queen's row
        if r == row:                           # same row → conflict
            return False
        if abs(r - row) == abs(c - col):     # same diagonal → conflict
            return False
    return True   # no conflict found → safe to place

# goal_fn: We are done when all 8 columns have a queen placed.
# state is a tuple of row positions — e.g. (2, 0, 5, 7, 1, 4, 6, 3).
# Because is_safe() blocks conflicts during placement,
# a full board is automatically a valid solution.
def goal_fn(state):
    return len(state) == N   # True when all 8 queens are placed

# expand_fn: Place a queen in the NEXT empty column.
# The next column index = len(state) (we fill columns left to right).
# e.g. state=(2,0) → 2 queens placed → next column is col 2.
# Try every row in that column — only include rows that pass is_safe().
def expand_fn(node):
    placement = node.state
    col = len(placement)    # next column to fill
    if col >= N:
        return []           # board is full, no more moves
    children = []
    for row in range(N):                          # try each row in this column
        if is_safe(placement, col, row):
            new_state = placement + (row,)        # append new queen's row to the tuple
            step_cost = row + 1                  # cost (used by UCS to prefer lower rows)
            children.append((new_state, step_cost))
    return children

# h_fn: Count attacking pairs among queens already placed.
# h = 0  → no conflicts — perfect so far (or goal reached).
# h > 0  → some queens are attacking each other.
# Two queens attack if they are in the same ROW or same DIAGONAL.
# (Same column is impossible — we place exactly one per column.)
def h_fn(state):
    conflicts = 0
    for col_a in range(len(state)):
        for col_b in range(col_a+1, len(state)):
            row_a = state[col_a]   # row of queen in column col_a
            row_b = state[col_b]   # row of queen in column col_b
            same_row      = (row_a == row_b)
            same_diagonal = (abs(row_a - row_b) == abs(col_a - col_b))
            if same_row or same_diagonal:
                conflicts += 1
    return conflicts   # lower = better; 0 = no conflicts

def print_board(placement):
    """Pretty-print the chessboard."""
    for row in range(N):
        print(' '.join('♛' if placement[col]==row else '·' for col in range(N)))
    print()

# ── Run ──────────────────────────────────────────────────────

    print(f"── {algo.upper()} (expanded={result['expanded']}) ──")
    if sol: print_board(sol)
    else: print("No solution\n")
Problem formulation
State: a tuple of 9 integers representing the 3×3 board, left-to-right top-to-bottom. 0 is the blank tile. E.g. (1,2,3,4,0,6,7,5,8).
Initial state: any solvable permutation of 0–8.
Actions: slide a tile into the blank — equivalent to moving the blank Up, Down, Left, or Right.
Step cost: 1 per move (uniform).
Goal state: (1,2,3,4,5,6,7,8,0) — blank in bottom-right.
Goal test: state == GOAL.
Heuristic (Best-First / A*): sum of Manhattan distances of each tile from its current position to its goal position. This is admissible because each tile needs at least that many moves to reach its goal, and tiles cannot pass through each other.

# ── Problem definition ────────────────────────────────────────
GOAL = (1,2,3, 4,5,6, 7,8,0)
# Position of each tile value in the goal state (for heuristic)
GOAL_POS = {v: (divmod(i,3)) for i,v in enumerate(GOAL)}

# goal_fn: The puzzle is solved when every tile is in the correct position.
# state is a flat 9-tuple — compare directly to the GOAL tuple.
def goal_fn(state):
    return state == GOAL   # e.g. (1,2,3,4,5,6,7,8,0)

# expand_fn: Move the blank tile (0) in each of 4 directions.
# "Moving the blank UP" = the tile above slides DOWN into the blank.
# Each move produces one new board state at cost = 1.
def expand_fn(node):
    state  = list(node.state)     # convert tuple to list for swapping
    z      = state.index(0)       # flat index of the blank (0)
    zr, zc = divmod(z, 3)         # convert flat index → (row, col)
    moves  = []
    for dr, dc in [(-1,0),(1,0),(0,-1),(0,1)]:   # Up, Down, Left, Right
        nr, nc = zr+dr, zc+dc
        if 0 <= nr < 3 and 0 <= nc < 3:        # within the 3×3 board
            ni = nr*3 + nc             # flat index of the tile to slide
            new_state     = state[:]    # copy the board
            new_state[z]  = new_state[ni]  # tile slides into blank position
            new_state[ni] = 0             # blank moves to tile's old position
            moves.append((tuple(new_state), 1))   # cost = 1 per slide
    return moves

# h_fn: Sum of Manhattan distances for every non-blank tile.
# For each tile, count how many rows + columns it is away from where it should be.
# ADMISSIBLE: a tile needs AT LEAST this many moves to reach its goal position.
def h_fn(state):
    total = 0
    for i, tile in enumerate(state):
        if tile == 0:
            continue                   # skip the blank — it has no goal
        cr, cc = divmod(i, 3)          # current (row, col) of this tile
        gr, gc = GOAL_POS[tile]         # goal    (row, col) of this tile
        total += abs(cr-gr) + abs(cc-gc)   # Manhattan distance for this tile
    return total   # 0 = every tile is already in its correct position

def shuffle(n_moves=30):
    """Create a random solvable puzzle by making n random moves from GOAL."""
    state = list(GOAL)
    for _ in range(n_moves):
        z  = state.index(0); zr, zc = divmod(z, 3)
        nb = [(zr+dr,zc+dc) for dr,dc in [(-1,0),(1,0),(0,-1),(0,1)] if 0<=zr+dr<3 and 0<=zc+dc<3]
        nr, nc = random.choice(nb); ni = nr*3+nc
        state[z], state[ni] = state[ni], state[z]
    return tuple(state)

def print_board(state):
    for i in range(3):
        print(' '.join('_' if v==0 else str(v) for v in state[i*3:i*3+3]))
    print()

# ── Run ──────────────────────────────────────────────────────
START = shuffle(20)   # random 20-move scramble
print("Initial state:"); print_board(START)

for algo in ['bfs', 'ucs', 'bestfs', 'astar']:
    result = search(START, goal_fn, expand_fn, h_fn, algo)
    path   = result['path']
    print(f"{algo.upper():8} moves={len(path)-1 if path else '✗'}  expanded={result['expanded']}")
# DFS with depth-limit=31 to avoid infinite loops:
result = search(START, goal_fn, expand_fn, h_fn, 'dfs', depth_limit=31)
print(f"DFS(d=31) moves={len(result['path'])-1 if result['path'] else '✗'}  expanded={result['expanded']}")
Why DFS needs a depth limit: the 8-puzzle has 9! = 362,880 states. Without a limit DFS can explore billions of nodes by revisiting permutations in a different order. A* with Manhattan distance typically solves any 20-move scramble in under 1,000 expansions.
